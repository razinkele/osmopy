```python
def _state_species_id(
    species_id: list[int], abundance: list[float] | None = None
) -> SchoolState:
    """Build a SchoolState with given species_id and (optional) per-school abundance.

    Defaults to unit abundance so existing equal-weight assertions hold. weight is set
    to 1.0 and biomass = abundance * weight to satisfy SchoolState's biomass invariant.
    """
    n = len(species_id)
    sp = np.array(species_id, dtype=np.int32)
    ab = np.ones(n, dtype=np.float64) if abundance is None else np.array(abundance, dtype=np.float64)
    state = SchoolState.create(n_schools=n, species_id=sp)
    state = state.replace(abundance=ab, weight=np.ones(n, dtype=np.float64), biomass=ab.copy())
    return state
```

```python
def test_collect_trait_stats_weights_by_abundance() -> None:
    """A huge cohort must dominate the mean over a tiny cohort."""
    state = _state_species_id([0, 0], abundance=[1.0, 999.0])
    out = _collect_trait_stats(state, {"imax": np.array([2.0, 10.0])})
    # abundance-weighted mean ~= 10.0 (the 999-abundance school dominates),
    # NOT the unweighted 6.0.
    assert out["imax"][0].mean == pytest.approx((2.0 * 1.0 + 10.0 * 999.0) / 1000.0)
```
