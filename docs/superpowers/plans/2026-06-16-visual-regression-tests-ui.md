# Visual Regression Tests for the UI — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add Playwright screenshot visual-regression tests for the four deterministic config page bodies (Species/Setup, Fishing, Movement, Advanced), with a pure numpy/Pillow comparison core, an opt-in containerized CI gate, and CI-generated baselines.

**Architecture:** Snapshots are **clipped to each page's `#split_<page>` body element** (which structurally excludes the live header chrome and the Setup validation panel). A per-test `prepare_page` loads the deterministic `minimal` demo and pins theme/animations/focus. A pure `compare_images(numpy+Pillow)` core (unit-tested in the normal suite) does pixel diffing. Browser tests are marked `visual`, guarded out of normal CI collection, and run only via a pinned Playwright container in an opt-in `visual.yml` workflow; baselines are generated by a `workflow_dispatch` job and delivered as an artifact the developer commits.

**Tech Stack:** Python 3.12, pytest, pytest-playwright (sync API) 0.7.2, playwright 1.58.0, Pillow, numpy, GitHub Actions, `mcr.microsoft.com/playwright/python` container.

**Spec:** `docs/superpowers/specs/2026-06-16-visual-regression-tests-ui-design.md` (converged through 3 in-loop review rounds).

**Branch:** `feat/visual-regression-tests` (holds the spec + this plan). Do NOT start on master.

```python
no browser). Pillow is in [dev]; the find_spec guard is defensive."""

from __future__ import annotations

import io
from importlib.util import find_spec

import numpy as np
import pytest

if find_spec("PIL") is None:  # pragma: no cover - Pillow is in [dev]
    pytest.skip("Pillow not installed", allow_module_level=True)

from PIL import Image

from tests._visual_compare import compare_images


def _png(arr: np.ndarray) -> bytes:
    buf = io.BytesIO()
    Image.fromarray(arr.astype("uint8"), mode="RGB").save(buf, format="PNG")
    return buf.getvalue()


def _solid(h: int, w: int, color=(10, 20, 30)) -> np.ndarray:
    a = np.zeros((h, w, 3), dtype="uint8")
    a[:, :] = color
    return a


def test_identical_images_pass():
    png = _png(_solid(20, 20))
    passed, ratio, _ = compare_images(png, png)
    assert passed is True
    assert ratio == 0.0


def test_subthreshold_noise_passes():
    base = _solid(20, 20, (100, 100, 100))
    act = base.copy()
    act[:, :] = (102, 100, 100)  # per-channel delta 2 < default threshold 4
    passed, ratio, _ = compare_images(_png(base), _png(act))
    assert passed is True
    assert ratio == 0.0


def test_dimension_mismatch_fails():
    passed, ratio, _ = compare_images(_png(_solid(20, 20)), _png(_solid(21, 20)))
    assert passed is False
    assert ratio == 1.0


def test_large_change_fails_and_highlights_red():
    base = _solid(20, 20, (0, 0, 0))
    act = base.copy()
    act[0:10, :] = (255, 255, 255)  # top half changes; 50% of pixels, delta 255
    passed, ratio, diff_png = compare_images(_png(base), _png(act), max_ratio=0.001)
    assert passed is False
    assert ratio > 0.4
    diff = np.asarray(Image.open(io.BytesIO(diff_png)).convert("RGB"))
    # changed region highlighted pure red
    assert (diff[0:10] == (255, 0, 0)).all()


def test_max_ratio_boundary():
    base = _solid(10, 10, (0, 0, 0))  # 100 px total
    act = base.copy()
    act[0, 0] = (255, 255, 255)  # 1 changed px = 1%
    assert compare_images(_png(base), _png(act), max_ratio=0.02)[0] is True   # 1% <= 2%
    assert compare_images(_png(base), _png(act), max_ratio=0.005)[0] is False  # 1% > 0.5%
```

```python
"""Pure image-comparison core for visual-regression tests.

numpy + Pillow only — NO playwright import, so the unit tests can run in the normal
CI suite. The browser harness (tests/_visual_support.py) imports this.
"""

from __future__ import annotations

import io

import numpy as np
from PIL import Image


def _to_rgb_array(png_bytes: bytes) -> np.ndarray:
    with Image.open(io.BytesIO(png_bytes)) as im:
        return np.asarray(im.convert("RGB"), dtype=np.int16)


def compare_images(
    baseline_png: bytes,
    actual_png: bytes,
    *,
    threshold: int = 4,
    max_ratio: float = 0.001,
) -> tuple[bool, float, bytes]:
    """Compare two PNG byte strings.

    A pixel "differs" when its max per-channel absolute difference exceeds ``threshold``
    (absorbs sub-pixel antialiasing). Returns ``(passed, diff_ratio, diff_png)``:
    - ``passed`` is False if dimensions differ, or if the fraction of differing pixels
      exceeds ``max_ratio``.
    - ``diff_ratio`` is that fraction (1.0 on a dimension mismatch).
    - ``diff_png`` highlights differing pixels in red (#FF0000) over a dimmed baseline;
      on a dimension mismatch it is the actual image unchanged.
    """
    base = _to_rgb_array(baseline_png)
    act = _to_rgb_array(actual_png)
    if base.shape != act.shape:
        return (False, 1.0, actual_png)
    per_pixel_delta = np.abs(base - act).max(axis=2)
    differing = per_pixel_delta > threshold
    ratio = float(differing.mean())
    highlight = (base // 3).astype(np.uint8)  # dimmed baseline backdrop
    highlight[differing] = (255, 0, 0)
    buf = io.BytesIO()
    Image.fromarray(highlight, mode="RGB").save(buf, format="PNG")
    return (ratio <= max_ratio, ratio, buf.getvalue())
```

```python
"""Playwright helpers for visual-regression snapshots.

Imports playwright (browser-only) — guarded out of normal CI collection by
tests/conftest.py. The pure pixel diff lives in tests/_visual_compare.py.
"""

from __future__ import annotations

import os
from pathlib import Path

from playwright.sync_api import Page, expect

from tests._e2e_support import dismiss_changelog_modal
from tests._visual_compare import compare_images

_TIMEOUT = 20_000
_BASELINE_DIR = Path(__file__).parent / "visual_baselines"
_OUTPUT_DIR = Path(__file__).parent / "visual_output"

# Kill animation/focus/scrollbar nondeterminism inside the captured element.
_DETERMINISM_CSS = (
    "*{transition:none!important;animation:none!important;caret-color:transparent!important}"
    "*:focus{outline:none!important}"
    "::-webkit-scrollbar{display:none!important}"
)

_NAV_TO_CLIP = {
    "setup": "#split_setup",
    "fishing": "#split_fishing",
    "movement": "#split_movement",
    "advanced": "#split_advanced",
}


def _update_mode() -> bool:
    return os.environ.get("OSMOSE_UPDATE_SNAPSHOTS", "").lower() in ("1", "true", "yes")


def _max_ratio(default: float) -> float:
    # Local advisory runs (native browser) drift on font AA vs the container; allow a
    # looser ratio via env without touching the committed default used by the CI gate.
    env = os.environ.get("OSMOSE_VISUAL_MAX_RATIO")
    return float(env) if env else default


def prepare_page(page: Page, app_url: str) -> None:
    """Load the app to a deterministic state with the `minimal` demo loaded.

    Pins theme=light before connect, dismisses the startup changelog modal, loads the
    `minimal` demo from the Domain page, and gates on the header config-stats (which read
    server-side state directly — no client round-trip) before returning.
    """
    page.add_init_script("try { localStorage.setItem('osmose-theme', 'light'); } catch (e) {}")
    page.set_viewport_size({"width": 1280, "height": 900})
    page.goto(app_url)
    page.wait_for_selector(".nav-pills", timeout=_TIMEOUT)
    dismiss_changelog_modal(page)

    # Load the deterministic `minimal` demo (controls live only on the Domain page).
    page.locator(".nav-pills .nav-link[data-value='grid']").click()
    page.wait_for_selector("#load_example", timeout=_TIMEOUT)
    page.select_option("#load_example", "minimal")
    page.click("#btn_load_example")
    # config_header renders ".osm-config-stats" only once a config is loaded (app.py:531).
    expect(page.locator(".osm-config-stats")).to_contain_text("params", timeout=_TIMEOUT)

    page.add_style_tag(content=_DETERMINISM_CSS)
    page.evaluate("async () => { await document.fonts.ready; }")


def navigate_to(page: Page, nav_value: str) -> str:
    """Navigate to a config page by nav value; return its clip selector once content rendered."""
    clip = _NAV_TO_CLIP[nav_value]
    page.locator(f".nav-pills .nav-link[data-value='{nav_value}']").click()
    page.wait_for_selector(clip, state="visible", timeout=_TIMEOUT)
    # Nav panels render lazily on shown.bs.tab (app.py:173) -- wait for inner content.
    page.wait_for_selector(f"{clip} .card", timeout=_TIMEOUT)
    page.mouse.move(0, 0)
    page.evaluate("() => document.activeElement && document.activeElement.blur()")
    expect(page.locator(".tooltip, .popover")).to_have_count(0)
    return clip


def assert_clip_snapshot(
    page: Page,
    clip_selector: str,
    name: str,
    *,
    mask=None,
    threshold: int = 4,
    max_ratio: float = 0.01,
) -> None:
    """Screenshot the clip element and compare to the committed baseline.

    In OSMOSE_UPDATE_SNAPSHOTS mode, (re)writes the baseline and returns. Otherwise
    compares; on failure writes <name>.actual.png + <name>.diff.png to tests/visual_output/.
    """
    shot = page.locator(clip_selector).screenshot(mask=mask or [], mask_color="#FF00FF")
    baseline = _BASELINE_DIR / f"{name}.png"
    if _update_mode():
        _BASELINE_DIR.mkdir(parents=True, exist_ok=True)
        baseline.write_bytes(shot)
        return
    assert baseline.exists(), (
        f"No baseline for {name!r} at {baseline}. Generate it via the visual-update CI "
        f"job (or locally with OSMOSE_UPDATE_SNAPSHOTS=1 pytest -m visual)."
    )
    passed, ratio, diff_png = compare_images(
        baseline.read_bytes(), shot, threshold=threshold, max_ratio=_max_ratio(max_ratio)
    )
    if not passed:
        _OUTPUT_DIR.mkdir(parents=True, exist_ok=True)
        (_OUTPUT_DIR / f"{name}.actual.png").write_bytes(shot)
        (_OUTPUT_DIR / f"{name}.diff.png").write_bytes(diff_png)
        raise AssertionError(
            f"Visual snapshot {name!r} differs: {ratio:.4%} of pixels exceed threshold "
            f"(max_ratio={_max_ratio(max_ratio):.4%}). See tests/visual_output/{name}.diff.png"
        )
```

```yaml
name: Visual

# Opt-in: the gate runs only on PRs that touch UI-affecting paths, plus manual dispatch.
# It is ADVISORY -- do NOT add `visual-gate` to branch-protection required checks (a
# path-filtered job reports no status on unrelated PRs and would deadlock merges).
on:
  pull_request:
    paths:
      - "ui/**"
      - "www/**"
      - "app.py"
  workflow_dispatch:
    inputs:
      ref:
        description: "Branch/ref to (re)generate baselines against"
        required: false

jobs:
  visual-gate:
    if: github.event_name == 'pull_request' || github.event_name == 'workflow_dispatch'
    runs-on: ubuntu-latest
    # Pinned Playwright image: bundled browsers + fonts match the playwright wheel
    # pinned in pyproject [viztest] (playwright==1.58.0). Keep this tag and that pin
    # in lockstep; the version-assert step below fails loudly on drift.
    container: mcr.microsoft.com/playwright/python:v1.58.0-noble
    steps:
      - uses: actions/checkout@v5
      - name: Trust the workspace (container runs as root vs checkout UID)
        run: git config --global --add safe.directory "$GITHUB_WORKSPACE"
      - name: Install package with visual-test extra
        run: pip install -e ".[viztest]"
      - name: Assert playwright wheel matches the container image
        run: |
          python - <<'PY'
          import sys
          from importlib.metadata import version
          installed = version("playwright")
          expected = "1.58.0"  # must equal the container image tag above
          if installed != expected:
              sys.exit(f"playwright {installed} != image-pinned {expected}; update the image tag and the [viztest] pin together")
          print("playwright", installed, "matches image")
          PY
      - name: Run visual snapshots
        run: python -m pytest -m visual
      - name: Upload diff artifacts on failure
        if: failure()
        uses: actions/upload-artifact@v4
        with:
          name: visual-diffs
          path: tests/visual_output/
          retention-days: 14
          if-no-files-found: ignore

  visual-update:
    # Manual only: regenerate baselines in the container and hand them back as an
    # artifact for a developer to download and commit (no contents:write needed; works
    # regardless of fork-PR token restrictions). Dispatch from the PR's head branch.
    if: github.event_name == 'workflow_dispatch'
    runs-on: ubuntu-latest
    container: mcr.microsoft.com/playwright/python:v1.58.0-noble
    steps:
      - uses: actions/checkout@v5
        with:
          ref: ${{ github.event.inputs.ref || github.ref }}
      - name: Trust the workspace
        run: git config --global --add safe.directory "$GITHUB_WORKSPACE"
      - name: Install package with visual-test extra
        run: pip install -e ".[viztest]"
      - name: Regenerate baselines
        env:
          OSMOSE_UPDATE_SNAPSHOTS: "1"
        run: python -m pytest -m visual
      - name: Upload regenerated baselines
        uses: actions/upload-artifact@v4
        with:
          name: visual-baselines
          path: tests/visual_baselines/*.png
          retention-days: 30
          if-no-files-found: error
```
