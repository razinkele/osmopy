> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Bundle the OSMOSE 4.4.0 jar, make OSMOPY write native 4.4.0 configs with value migrations that preserve engine behavior (the 4.4.0 larval-mortality units change), validate via A/B against the 4.3.3 jar, then hard-replace the jar.

**Architecture:** A value-migration layer in `osmose/config/aliases.py` scales the per-year/per-timestep larval-rate keys at the I/O boundary (×ndt writing native 4.4.0, ÷ndt reading a native-4.4.0 config), keeping the internal model in 4.3.x/per-timestep units (what the Python engine consumes). Phases: A = value layer (CI-testable), B = fetch jar + A/B validation gate (local), C = cutover (flip write-default, remove old jar, version-aware glob, refs), D = deploy guard + docs.

**Tech Stack:** Python 3.12, pytest, OSMOSE Java jar (subprocess). Run with `.venv/bin/python`.

**Spec:** `docs/superpowers/specs/2026-06-18-jar-swap-4.4.0-design.md` (approved). Built on PR1 (`001c530`) + PR2 (`dc2d701`).

```python
import re
import warnings

# Larval additional-mortality SCALAR per-species rate. 4.4.0 reinterprets this as rate/YEAR
# (AnnualLarvaMortality.init() adds `/ nStepYear`); 4.3.x and the OSMOPY internal model treat
# it as rate/time-step. So multiply by ndtperyear when writing native 4.4.0, divide when reading
# a native-4.4.0 config. Matches ONLY `...rate.spN` (NOT the `.bydt.file.spN` path variant nor
# `.seasonality.file.spN`).
_LARVA_RATE_RE = re.compile(r"^mortality\.additional\.larva\.rate\.sp\d+$")
_NDT_KEY = "simulation.time.ndtperyear"


def _ndtperyear(cfg: dict[str, str]) -> float | None:
    raw = cfg.get(_NDT_KEY)
    if raw in (None, ""):
        return None
    try:
        return float(raw)
    except ValueError:
        return None


def _scale_rate_value(value: str, factor: float) -> str:
    # ';'-separated per-stage arrays are scaled component-wise; :.10g trims float noise.
    return ";".join(f"{float(p) * factor:.10g}" for p in value.split(";"))


def _migrate_larva_rate(cfg: dict[str, str], factor: float, *, warn_bydt: bool) -> dict[str, str]:
    """Scale per-year larval-rate scalars by `factor` (ndt to write 4.4.0, 1/ndt to read it)."""
    result = dict(cfg)
    has_larva = any(_LARVA_RATE_RE.match(k) for k in result)
    if has_larva and _ndtperyear(result) is None:
        warnings.warn(
            "larval additional-mortality rate present but simulation.time.ndtperyear is missing; "
            "skipping the 4.4.0 per-year unit migration (config may be mis-scaled for the jar).",
            stacklevel=2,
        )
        return result
    for key, value in list(result.items()):
        if _LARVA_RATE_RE.match(key):
            result[key] = _scale_rate_value(value, factor)
        elif warn_bydt and key.startswith("mortality.additional.larva.rate.bydt.file"):
            warnings.warn(
                f"{key} references a per-time-step larval-rate file that 4.4.0 reads as rate/year; "
                "OSMOPY does not rescale the referenced file — verify it manually.",
                stacklevel=2,
            )
    return result
```

```python
def canonicalize_config(cfg: dict[str, str]) -> tuple[dict[str, str], list[str]]:
    from osmose.demo import migrate_config, _version_tuple

    deprecated = sorted(
        k for k in cfg if any(k == old or k.startswith(old + ".") for old in RENAMES_440)
    )
    migrated = migrate_config(cfg, target_version="4.4.0")
    if _version_tuple(cfg.get("osmose.version", "")) >= _version_tuple("4.4.0"):
        ndt = _ndtperyear(migrated)
        if ndt:
            migrated = _migrate_larva_rate(migrated, 1.0 / ndt, warn_bydt=False)
    return migrated, deprecated
```

```python
def _drop_4_4_0_removed_keys(cfg: dict[str, str]) -> dict[str, str]:
    """Drop keys that 4.4.0 removed or reinterprets, to preserve legacy behavior on write."""
    result = dict(cfg)
    # 4.4.0 removed the species.lmax growth cap (no clean migration; documented limitation).
    for key in [k for k in result if k == "species.lmax" or k.startswith("species.lmax.")]:
        del result[key]
    # 4.4.0 species.beta feeds the predation allometric exponent (default 1 == legacy). 4.3.3 read
    # it only under bioen; for a non-bioen config it must not be emitted.
    bioen_on = str(result.get("module.bioenergetics.enabled", "false")).lower() == "true"
    if not bioen_on:
        for key in [k for k in result if k == "species.beta" or k.startswith("species.beta.")]:
            del result[key]
    return result
```

```python
import re as _re

def _pick_default_jar(jars: list[Path]) -> Path | None:
    if not jars:
        return None
    def _ver(p: Path) -> tuple[int, ...]:
        m = _re.search(r"osmose_(\d+)\.(\d+)\.(\d+)", p.name)
        return tuple(int(g) for g in m.groups()) if m else (0,)
    return max(jars, key=lambda p: (_ver(p), p.name))
```
